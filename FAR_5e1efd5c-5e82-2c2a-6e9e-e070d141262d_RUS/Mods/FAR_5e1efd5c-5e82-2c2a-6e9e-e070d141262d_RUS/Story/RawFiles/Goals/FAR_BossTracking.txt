PROC
PROC_FAR_SyncBossUI()
AND
GetHostCharacter((CHARACTER)_Host)
AND
DB_FAR_BossKillCount(_Count)
AND
IntegerToReal(_Count, _CountReal)
AND
RealSubtract(0.0, 9999.0, _Floor)
THEN
PartyIncreaseActionResourceValue(_Host, "FAR_BossCount", _Floor);
PartyIncreaseActionResourceValue(_Host, "FAR_BossCount", _CountReal);
PROC_FAR_SyncUIVisibility(_Host);

PROC
PROC_FAR_SyncUIVisibility((CHARACTER)_Host)
AND
DB_FAR_UIEnabled(1)
THEN
PROC_FAR_SetUIVisibility(_Host, 1);

PROC
PROC_FAR_SyncUIVisibility((CHARACTER)_Host)
AND
NOT DB_FAR_UIEnabled(1)
THEN
PROC_FAR_SetUIVisibility(_Host, 0);

PROC
PROC_FAR_SetUIVisibility((CHARACTER)_Host, (INTEGER)_State)
AND
IntegerToReal(_State, _StateReal)
AND
RealSubtract(0.0, 9999.0, _Floor)
THEN
PartyIncreaseActionResourceValue(_Host, "FAR_BossCount_Visible", _Floor);
PartyIncreaseActionResourceValue(_Host, "FAR_BossCount_Visible", _StateReal);

// Unlocking Legendary auto-disables the boss counter
PROC
PROC_FAR_EvaluateCampSequence((CHARACTER)_Character)
AND
DB_FAR_BossKillCount(_Count)
AND
_Count >= 30
AND
NOT DB_FAR_MessageShown_Tier(30)
AND
GetHostCharacter(_Host)
THEN
NOT DB_FAR_UIEnabled(1);
PROC_FAR_SetUIVisibility(_Host, 0);
ShowNotification(_Host, "Attunement complete. Boss Counter disabled.");
OpenMessageBox(_Character, "Ваше посвящение в Древнее Плетение завершено. Легендарные руны усиления открыты для использования и покупки.");
PROC_FAR_ApplyUnlockToParty("FAR_UNCOMMON_UNLOCKED");
PROC_FAR_ApplyUnlockToParty("FAR_RARE_UNLOCKED");
PROC_FAR_ApplyUnlockToParty("FAR_VERYRARE_UNLOCKED");
PROC_FAR_ApplyUnlockToParty("FAR_LEGENDARY_UNLOCKED");
DB_FAR_MessageShown_Tier(3);
DB_FAR_MessageShown_Tier(10);
DB_FAR_MessageShown_Tier(20);
DB_FAR_MessageShown_Tier(30);
